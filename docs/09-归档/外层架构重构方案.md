## 1. 背景与目标

### 1.1 当前状态

协议层已完成模块化拆分，建立了三个独立的协议模块：

```
src/protocol/
  ├── ssn_protocol.h/c      # 公共协议基础
  ├── rpc/
  │   ├── ssn_rpc.h         # RPC协议接口 (req/rep)
  │   └── ssn_rpc.c
  ├── pubsub/
  │   ├── ssn_pubsub.h      # 发布订阅协议接口 (pub/sub)
  │   └── ssn_pubsub.c
  └── msg/
      ├── ssn_msg.h         # 消息协议接口 (send/recv)
      └── ssn_msg.c
```

### 1.2 需要重构的外层文件

```
ssn_client.c    # IPC客户端实现 (78+ 行)
ssn_server.c    # IPC服务端实现
ssn_cliauto.c   # 客户端自动重连模块
```

### 1.3 重构目标

1. **解耦协议逻辑**：将协议处理逻辑下沉到协议层模块
2. **简化客户端/服务端**：客户端/服务端专注于连接管理和事件分发
3. **保持向后兼容**：确保现有API接口不变
4. **提升可维护性**：各模块职责清晰，便于维护和扩展

### 2.2 模块职责划分

#### ssn_client.c 职责
- **连接生命周期管理**：创建连接、断开连接、心跳检测
- **事件循环**：poll/run 机制
- **消息路由**：将协议层的消息路由到正确的回调
- **引用计数管理**：资源管理
- **全局客户端列表维护**

#### ssn_server.c 职责
- **服务器监听**：绑定地址、监听连接
- **客户端连接管理**：接受连接、维护客户端状态
- **事件循环**：poll/run 机制
- **消息路由**：将协议层的消息路由到正确的处理函数
- **发布订阅管理**：维护订阅者列表

#### ssn_cliauto.c 职责
- **自动重连逻辑**：连接失败后自动重试
- **自动订阅**：连接成功后自动订阅指定URL
- **状态机管理**：管理连接状态转换

### 2.3 协议层模块职责

#### ssn_rpc 模块
- **RPC请求管理**：序列号生成、待处理请求池
- **RPC响应匹配**：根据序列号匹配请求和响应
- **方法注册表**：服务端方法管理（供server层调用）

#### ssn_pubsub 模块
- **订阅者管理**：订阅/取消订阅
- **主题匹配**：消息发布时匹配订阅者
- **发布消息队列**：缓存待发布消息

#### ssn_msg 模块
- **消息队列管理**：send/recv 队列
- **消息分发**：将消息路由到正确的接收者

#### 3.1.2 客户端初始化流程（重构后）

```c
/* ssn_client.c 重构 - 客户端创建 */
ssn_client_t *ssn_client_create(ssn_client_msg_handler_t on_publish, void *arg)
{
    ssn_client_t *client = (ssn_client_t *)calloc(1, sizeof(ssn_client_t));
    if (!client) {
        return NULL;
    }

    /* 创建协议层实例 */
    client->rpc_req = ssn_rpc_req_create(rpc_reply_callback, client);
    if (!client->rpc_req) {
        free(client);
        return NULL;
    }

    client->pubsub_sub = ssn_pubsub_sub_create(pubsub_message_callback, client);
    if (!client->pubsub_sub) {
        ssn_rpc_destroy((ssn_protocol_ctx_t *)client->rpc_req);
        free(client);
        return NULL;
    }

    client->msg_send = ssn_msg_send_create();
    if (!client->msg_send) {
        ssn_pubsub_destroy((ssn_protocol_ctx_t *)client->pubsub_sub);
        ssn_rpc_destroy((ssn_protocol_ctx_t *)client->rpc_req);
        free(client);
        return NULL;
    }

    /* 设置回调 */
    client->onmsg = on_publish;
    client->msg_arg = arg;

    return client;
}

/* RPC应答回调 - 协议层调用 */
static void rpc_reply_callback(uint16_t seqno, uint32_t status,
                               const void *data, size_t data_len, void *arg)
{
    ssn_client_t *client = (ssn_client_t *)arg;
    if (client && client->onrpc) {
        ssn_header_t hdr = {0};
        hdr.seqno = ipc_htons(seqno);
        ipc_set_status(&hdr, status);
        ssn_data_ref_t data_ref = {
            .data = (void *)data,
            .length = data_len
        };
        client->onrpc(client, &hdr, &data_ref, client->rpc_arg);
    }
}

/* 订阅消息回调 - 协议层调用 */
static void pubsub_message_callback(const char *topic, const void *data,
                                   size_t data_len, void *arg)
{
    ssn_client_t *client = (ssn_client_t *)arg;
    if (client && client->onmsg) {
        ssn_url_ref_t url_ref = {
            .url = (char *)topic,
            .url_len = strlen(topic)
        };
        ssn_data_ref_t data_ref = {
            .data = (void *)data,
            .length = data_len
        };
        client->onmsg(client, &url_ref, &data_ref, client->msg_arg);
    }
}
```

#### 3.1.3 客户端连接流程（重构后）

```c
/* 客户端连接 - 重构后 */
bool ssn_client_connect(ssn_client_t *client, const char* ipc_path,
                        const struct timespec *timeout)
{
    /* 创建传输层 */
    client->transport = ssn_transport_create(SSN_TRANSPORT_TCP, NULL);
    if (!client->transport) {
        return false;
    }

    /* 解析地址 */
    ssn_address_t addr;
    if (!ssn_address_parse(ipc_path, &addr)) {
        ssn_transport_destroy(client->transport);
        client->transport = NULL;
        return false;
    }

    /* 连接传输层 */
    if (!ssn_transport_connect(client->transport, &addr,
                               timeout ? timeout->tv_sec * 1000 : 5000)) {
        ssn_transport_destroy(client->transport);
        client->transport = NULL;
        return false;
    }

    /* 绑定协议层到传输层 */
    ssn_rpc_req_connect(client->rpc_req, client->transport);
    ssn_pubsub_sub_connect(client->pubsub_sub, client->transport);
    ssn_msg_send_connect(client->msg_send, client->transport);

    client->connected = true;
    return true;
}
```

#### 3.2.2 服务端初始化流程（重构后）

```c
/* ssn_server.c 重构 - 服务端创建 */
ssn_server_t *ssn_server_create(const char* server_info)
{
    ssn_server_t *server = (ssn_server_t *)calloc(1, sizeof(ssn_server_t));
    if (!server) {
        return NULL;
    }

    server->server_info = strdup(server_info);

    /* 创建协议层实例 */
    server->rpc_rep = ssn_rpc_rep_create(rpc_request_callback, server);
    if (!server->rpc_rep) {
        free(server->server_info);
        free(server);
        return NULL;
    }

    server->pubsub_pub = ssn_pubsub_pub_create();
    if (!server->pubsub_pub) {
        ssn_rpc_destroy((ssn_protocol_ctx_t *)server->rpc_rep);
        free(server->server_info);
        free(server);
        return NULL;
    }

    server->msg_recv = ssn_msg_recv_create(msg_recv_callback, server);
    if (!server->msg_recv) {
        ssn_pubsub_destroy((ssn_protocol_ctx_t *)server->pubsub_pub);
        ssn_rpc_destroy((ssn_protocol_ctx_t *)server->rpc_rep);
        free(server->server_info);
        free(server);
        return NULL;
    }

    return server;
}

/* 服务端启动 - 重构后 */
bool ssn_server_start(ssn_server_t *server)
{
    /* 创建传输层 */
    server->transport = ssn_transport_create(SSN_TRANSPORT_TCP, NULL);
    if (!server->transport) {
        return false;
    }

    /* 解析地址 */
    ssn_address_t addr;
    if (!ssn_address_parse(server->server_info, &addr)) {
        ssn_transport_destroy(server->transport);
        server->transport = NULL;
        return false;
    }

    /* 绑定并监听 */
    if (!ssn_transport_bind(server->transport, &addr)) {
        ssn_transport_destroy(server->transport);
        server->transport = NULL;
        return false;
    }

    if (!ssn_transport_listen(server->transport, 10)) {
        ssn_transport_destroy(server->transport);
        server->transport = NULL;
        return false;
    }

    /* 绑定协议层到传输层 */
    ssn_rpc_rep_bind(server->rpc_rep, server->transport);
    ssn_pubsub_pub_bind(server->pubsub_pub, server->transport);
    ssn_msg_recv_bind(server->msg_recv, server->transport);

    server->running = true;
    return true;
}
```

## 7. 注意事项

### 7.1 线程安全性

- 协议层本身**不是**线程安全的
- 客户端/服务端需要保证在多线程环境下的同步
- 回调函数可能在不同线程中被调用

### 7.2 内存管理

- 协议上下文创建后需要调用对应 destroy 函数释放
- 客户端/服务端销毁时需要先销毁协议层实例
- 回调函数返回后，数据内存可能失效

### 7.3 错误处理

- 协议层操作失败会返回错误码
- 客户端/服务端需要正确处理协议层错误
- 断开连接后需要重建协议层绑定

### 7.4 性能考虑

- 避免在回调函数中执行耗时操作
- 大量消息时考虑批量处理
- 合理设置超时时间
