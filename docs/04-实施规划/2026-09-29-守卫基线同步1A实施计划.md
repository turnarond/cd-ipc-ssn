def _is_acceptance_snapshot(root: Path, path: Path) -> bool:
    """发布验收报告是发布时点的事实快照：豁免事实类扫描；治理与链接检查仍生效。"""

        return False
    return (
        len(parts) >= 3
        and parts[0] == "docs"
        and parts[1] == "05-部署手册"
        and "发布验收报告" in path.name
    )
