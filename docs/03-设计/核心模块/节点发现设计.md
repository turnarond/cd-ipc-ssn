## 1. 概述

### 1.1 设计目标

节点发现机制旨在实现分布式环境中节点的自动发现、注册和状态管理，使得节点能够动态地发现网络中的其他节点并与之通信，无需手动配置。

### 1.2 设计原则

1. **自动化**：节点启动后自动发现网络中其他节点
2. **自组织**：无中心节点，节点间对等发现
3. **实时性**：节点状态变化能够快速传播
4. **容错性**：能够处理网络分区和节点故障

### 2.2 发现模式

1. **组播发现模式**：
   - 适用于小型局域网环境
   - 使用组播地址进行广播发现
   - 无需中心服务器

2. **目录服务模式**：
   - 适用于大型分布式系统
   - 节点向目录服务器注册
   - 其他节点从目录服务器查询

   > ⚠️ 未实现（2.0 中为可选规划）

3. **混合模式**：
   - 结合组播和目录服务
   - 初始使用组播发现目录服务器
   - 之后通过目录服务器进行发现

## 5. 发现算法实现

### 5.1 组播发现算法

```c
// 组播发现实现
typedef struct {
    char multicast_group[32];          // 组播组地址
    uint16_t multicast_port;           // 组播端口
    int multicast_socket;              // 组播套接字
    struct ip_mreq mreq;              // 组播成员结构
} multicast_discovery_t;

// 加入组播组
static bool multicast_join_group(multicast_discovery_t* md, const char* interface_ip) {
    // 创建UDP套接字
    md->multicast_socket = socket(AF_INET, SOCK_DGRAM, 0);
    if (md->multicast_socket < 0) {
        LOG_ERROR("Failed to create multicast socket: %s", strerror(errno));
        return false;
    }
    
    // 设置地址重用
    int optval = 1;
    setsockopt(md->multicast_socket, SOL_SOCKET, SO_REUSEADDR, &optval, sizeof(optval));
    
    // 绑定地址
    struct sockaddr_in local_addr;
    memset(&local_addr, 0, sizeof(local_addr));
    local_addr.sin_family = AF_INET;
    local_addr.sin_addr.s_addr = INADDR_ANY;
    local_addr.sin_port = md->multicast_port;
    
    if (bind(md->multicast_socket, (struct sockaddr*)&local_addr, sizeof(local_addr)) < 0) {
        LOG_ERROR("Failed to bind multicast socket: %s", strerror(errno));
        close(md->multicast_socket);
        return false;
    }
    
    // 设置组播TTL
    uint8_t ttl = 3;
    setsockopt(md->multicast_socket, IPPROTO_IP, IP_MULTICAST_TTL, &ttl, sizeof(ttl));
    
    // 加入组播组
    memset(&md->mreq, 0, sizeof(md->mreq));
    md->mreq.imr_multiaddr.s_addr = inet_addr(md->multicast_group);
    md->mreq.imr_interface.s_addr = interface_ip ? inet_addr(interface_ip) : INADDR_ANY;
    
    if (setsockopt(md->multicast_socket, IPPROTO_IP, IP_ADD_MEMBERSHIP, &md->mreq, sizeof(md->mreq)) < 0) {
        LOG_ERROR("Failed to join multicast group: %s", strerror(errno));
        close(md->multicast_socket);
        return false;
    }
    
    // 设置非阻塞模式
    int flags = fcntl(md->multicast_socket, F_GETFL, 0);
    fcntl(md->multicast_socket, F_SETFL, flags | O_NONBLOCK);
    
    LOG_INFO("Joined multicast group %s:%d", md->multicast_group, ntohs(md->multicast_port));
    return true;
}

// 发送组播发现消息
static bool multicast_send_announce(ssn_discovery_engine_t* engine, const ssn_discovery_announce_t* announce) {
    multicast_discovery_t* md = (multicast_discovery_t*)engine->impl_data;
    
    // 构建目标地址
    struct sockaddr_in dest_addr;
    memset(&dest_addr, 0, sizeof(dest_addr));
    dest_addr.sin_family = AF_INET;
    dest_addr.sin_addr.s_addr = inet_addr(md->multicast_group);
    dest_addr.sin_port = md->multicast_port;
    
    // 发送消息
    ssize_t sent = sendto(md->multicast_socket, announce, sizeof(ssn_discovery_announce_t), 0,
                         (struct sockaddr*)&dest_addr, sizeof(dest_addr));
    
    if (sent < 0) {
        LOG_ERROR("Failed to send multicast announce: %s", strerror(errno));
        return false;
    }
    
    LOG_DEBUG("Sent multicast announce for node %s", announce->node_id);
    return true;
}

// 接收组播消息
static void* multicast_receive_thread(void* arg) {
    ssn_discovery_engine_t* engine = (ssn_discovery_engine_t*)arg;
    multicast_discovery_t* md = (multicast_discovery_t*)engine->impl_data;
    
    uint8_t buffer[4096];
    struct sockaddr_in from_addr;
    socklen_t from_len = sizeof(from_addr);
    
    while (engine->running) {
        // 接收数据
        ssize_t received = recvfrom(md->multicast_socket, buffer, sizeof(buffer), 0,
                                   (struct sockaddr*)&from_addr, &from_len);
        
        if (received < 0) {
            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                // 非阻塞，没有数据
                usleep(10000); // 10ms
                continue;
            }
            LOG_ERROR("Multicast receive error: %s", strerror(errno));
            continue;
        }
        
        // 解析消息
        ssn_discovery_header_t* header = (ssn_discovery_header_t*)buffer;
        
        // 验证魔数
        if (header->magic != 0xDC) {
            LOG_WARN("Invalid discovery packet magic: 0x%02x", header->magic);
            continue;
        }
        
        // 处理消息
        switch (header->msg_type) {
            case SSN_DISCOVERY_ANNOUNCE:
                handle_announce_message(engine, (ssn_discovery_announce_t*)buffer, &from_addr);
                break;
                
            case SSN_DISCOVERY_HEARTBEAT:
                handle_heartbeat_message(engine, (ssn_discovery_heartbeat_t*)buffer, &from_addr);
                break;
                
            case SSN_DISCOVERY_GOODBYE:
                handle_goodbye_message(engine, (ssn_discovery_goodbye_t*)buffer, &from_addr);
                break;
                
            case SSN_DISCOVERY_QUERY:
                handle_query_message(engine, (ssn_discovery_query_t*)buffer, &from_addr);
                break;
                
            default:
                LOG_WARN("Unknown discovery message type: %d", header->msg_type);
                break;
        }
    }
    
    return NULL;
}
```

```c
// 目录服务客户端
typedef struct {
    char server_address[256];          // 目录服务器地址
    uint16_t server_port;               // 目录服务器端口
    ssn_transport_t* transport;         // 传输层连接
    
    // 缓存
    ssn_hash_table_t* cache;            // 节点缓存
    time_t last_sync;                   // 最后同步时间
    
    // 重试
    uint32_t retry_count;               // 重试次数
    uint32_t max_retries;              // 最大重试次数
} directory_client_t;

// 向目录服务注册节点
static bool directory_register_node(directory_client_t* client, const ssn_discovery_announce_t* announce) {
    // 构建注册请求
    ssn_directory_register_request_t request;
    memset(&request, 0, sizeof(request));
    
    request.header.magic = 0xDC;
    request.header.version = 1;
    request.header.msg_type = SSN_DIRECTORY_REGISTER;
    request.header.ttl = 1;
    
    strncpy(request.node_id, announce->node_id, sizeof(request.node_id) - 1);
    strncpy(request.node_type, announce->node_type, sizeof(request.node_type) - 1);
    request.capabilities = announce->capabilities;
    request.protocol_version = announce->protocol_version;
    
    // 发送注册请求
    ssize_t sent = ssn_transport_send(client->transport, &request, sizeof(request));
    if (sent < 0) {
        LOG_ERROR("Failed to send directory registration");
        return false;
    }
    
    // 等待响应
    ssn_directory_register_response_t response;
    ssize_t received = ssn_transport_recv(client->transport, &response, sizeof(response), 5000);
    
    if (received < 0 || response.result != 0) {
        LOG_ERROR("Directory registration failed");
        return false;
    }
    
    LOG_INFO("Successfully registered with directory server");
    return true;
}

// 从目录服务查询节点
static int directory_query_nodes(directory_client_t* client, const char* node_type_filter,
                                discovery_node_entry_t** results, int max_results) {
    // 构建查询请求
    ssn_directory_query_request_t request;
    memset(&request, 0, sizeof(request));
    
    request.header.magic = 0xDC;
    request.header.version = 1;
    request.header.msg_type = SSN_DIRECTORY_QUERY;
    request.header.ttl = 1;
    
    if (node_type_filter) {
        strncpy(request.node_type_filter, node_type_filter, sizeof(request.node_type_filter) - 1);
    }
    
    request.max_results = max_results;
    
    // 发送查询请求
    ssize_t sent = ssn_transport_send(client->transport, &request, sizeof(request));
    if (sent < 0) {
        LOG_ERROR("Failed to send directory query");
        return -1;
    }
    
    // 等待响应
    ssn_directory_query_response_t response;
    ssize_t received = ssn_transport_recv(client->transport, &response, sizeof(response), 5000);
    
    if (received < 0) {
        LOG_ERROR("Directory query failed");
        return -1;
    }
    
    // 解析结果
    int count = response.node_count < max_results ? response.node_count : max_results;
    for (int i = 0; i < count; i++) {
        memcpy(&results[i], &response.nodes[i], sizeof(discovery_node_entry_t));
    }
    
    return count;
}
```

### 5.3 心跳和故障检测

```c
// 心跳处理线程
static void* heartbeat_thread_func(void* arg) {
    ssn_discovery_engine_t* engine = (ssn_discovery_engine_t*)arg;
    
    while (engine->running) {
        // 睡眠直到下一次心跳
        usleep(engine->config.heartbeat_interval_ms * 1000);
        
        if (!engine->running) {
            break;
        }
        
        // 发送本地心跳
        ssn_discovery_heartbeat_t heartbeat;
        memset(&heartbeat, 0, sizeof(heartbeat));
        
        heartbeat.header.magic = 0xDC;
        heartbeat.header.version = 1;
        heartbeat.header.msg_type = SSN_DISCOVERY_HEARTBEAT;
        heartbeat.header.ttl = 2;
        
        strncpy(heartbeat.node_id, engine->local_node_id, sizeof(heartbeat.node_id) - 1);
        heartbeat.uptime_sec = get_uptime_seconds();
        heartbeat.active_connections = get_active_connection_count();
        heartbeat.cpu_usage = get_cpu_usage();
        heartbeat.memory_usage = get_memory_usage();
        
        // 发送心跳
        if (engine->config.enable_multicast) {
            multicast_send_heartbeat(engine, &heartbeat);
        }
        
        // 检查远程节点心跳
        check_remote_nodes_heartbeat(engine);
    }
    
    return NULL;
}

// 检查远程节点心跳
static void check_remote_nodes_heartbeat(ssn_discovery_engine_t* engine) {
    ipc_mutex_lock(engine->node_table_lock);
    
    // 遍历所有节点
    ssn_hash_table_iter_t iter;
    ssn_hash_table_iter_init(&iter, engine->node_table);
    
    time_t now = time(NULL);
    uint32_t timeout = engine->config.heartbeat_timeout_ms / 1000;
    
    while (ssn_hash_table_iter_next(&iter)) {
        discovery_node_entry_t* entry = (discovery_node_entry_t*)iter.value;
        
        if (entry->state == SSN_NODE_STATE_OFFLINE || 
            entry->state == SSN_NODE_STATE_REMOVED) {
            continue;
        }
        
        // 计算超时
        time_t elapsed = now - entry->last_heartbeat;
        
        if (elapsed > timeout) {
            entry->missed_heartbeats++;
            
            if (entry->missed_heartbeats >= 3) {
                // 节点疑似离线
                entry->state = SSN_NODE_STATE_SUSPECTED;
                
                LOG_WARN("Node %s missed %d heartbeats, marking as suspected",
                        entry->node_id, entry->missed_heartbeats);
                
                // 触发离线回调
                if (engine->callbacks.on_node_suspected) {
                    engine->callbacks.on_node_suspected(engine, entry, engine->callback_arg);
                }
            }
        }
    }
    
    ipc_mutex_unlock(engine->node_table_lock);
}

// 清理离线节点线程
static void* cleanup_thread_func(void* arg) {
    ssn_discovery_engine_t* engine = (ssn_discovery_engine_t*)arg;
    
    while (engine->running) {
        // 睡眠直到下一次清理
        usleep(engine->config.cleanup_interval_ms * 1000);
        
        if (!engine->running) {
            break;
        }
        
        cleanup_offline_nodes(engine);
    }
    
    return NULL;
}

// 清理离线节点
static void cleanup_offline_nodes(ssn_discovery_engine_t* engine) {
    ipc_mutex_lock(engine->node_table_lock);
    
    ssn_hash_table_iter_t iter;
    ssn_hash_table_iter_init(&iter, engine->node_table);
    
    time_t now = time(NULL);
    uint32_t cleanup_timeout = (engine->config.heartbeat_timeout_ms * 3) / 1000; // 3倍心跳超时
    
    ssn_list_t* to_remove = ssn_list_create();
    
    while (ssn_hash_table_iter_next(&iter)) {
        discovery_node_entry_t* entry = (discovery_node_entry_t*)iter.value;
        
        // 检查是否需要清理
        if (entry->state == SSN_NODE_STATE_SUSPECTED) {
            time_t elapsed = now - entry->last_heartbeat;
            if (elapsed > cleanup_timeout) {
                entry->state = SSN_NODE_STATE_OFFLINE;
                to_remove->append(to_remove, entry->node_id);
                
                LOG_INFO("Node %s is offline, removing from discovery table", entry->node_id);
                
                // 触发离线回调
                if (engine->callbacks.on_node_offline) {
                    engine->callbacks.on_node_offline(engine, entry, engine->callback_arg);
                }
            }
        }
    }
    
    ipc_mutex_unlock(engine->node_table_lock);
    
    // 实际删除节点
    ssn_list_node_t* node = to_remove->head;
    while (node) {
        char* node_id = (char*)node->data;
        ssn_hash_table_remove(engine->node_table, node_id);
        node = node->next;
    }
    
    ssn_list_destroy(to_remove);
}
```

```c
// 服务发现接口
typedef struct {
    // 注册服务
    bool (*register_service)(ssn_discovery_engine_t* engine, const ssn_service_info_t* service);
    
    // 注销服务
    bool (*unregister_service)(ssn_discovery_engine_t* engine, const char* service_id);

    // 查询服务

                         const char* service_name, ssn_service_info_t* results, int max_results);
    
    // 订阅服务变化
    bool (*subscribe_service_changes)(ssn_discovery_engine_t* engine,
                                    void (*callback)(ssn_discovery_engine_t* engine,
                                                   const char* service_id,
                                                   ssn_service_change_type_t change_type,
                                                   void* arg),
                                    void* arg);
} ssn_service_discovery_interface_t;

// 注册服务
static bool discovery_register_service(ssn_discovery_engine_t* engine, const ssn_service_info_t* service) {
    ipc_mutex_lock(engine->node_table_lock);
    
    // 查找提供服务的节点
    discovery_node_entry_t* node_entry = (discovery_node_entry_t*)ssn_hash_table_get(
        engine->node_table, service->node_id);
    
    if (!node_entry) {
        LOG_ERROR("Cannot register service: node %s not found", service->node_id);
        ipc_mutex_unlock(engine->node_table_lock);
        return false;
    }
    
    // 添加服务到节点
    if (node_entry->service_count >= MAX_SERVICES_PER_NODE) {
        LOG_ERROR("Node %s has reached maximum service limit", service->node_id);
        ipc_mutex_unlock(engine->node_table_lock);
        return false;
    }
    
    // 复制服务信息
    memcpy(&node_entry->services[node_entry->service_count], service, sizeof(ssn_service_info_t));
    node_entry->service_count++;
    
    // 添加到服务表
    char service_key[128];
    snprintf(service_key, sizeof(service_key), "%s:%s", service->service_type, service->service_name);
    
    ssn_hash_table_set(engine->service_table, strdup(service_key), &node_entry->services[node_entry->service_count - 1]);
    
    ipc_mutex_unlock(engine->node_table_lock);
    
    // 广播服务更新
    if (engine->config.enable_multicast) {
        broadcast_service_update(engine, service, SSN_SERVICE_ADDED);
    }
    
    LOG_INFO("Service %s registered on node %s", service->service_name, service->node_id);
    return true;
}

// 查询服务
static int discovery_query_services(ssn_discovery_engine_t* engine, const char* service_type,
                                  const char* service_name, ssn_service_info_t* results, int max_results) {
    ipc_mutex_lock(engine->node_table_lock);
    
    int result_count = 0;
    
    // 构建查询键
    char service_key[128];
    if (service_type && service_name) {
        snprintf(service_key, sizeof(service_key), "%s:%s", service_type, service_name);
    } else if (service_type) {
        snprintf(service_key, sizeof(service_key), "%s:", service_type);
    } else {
        service_key[0] = '\0';
    }
    
    // 遍历服务表
    ssn_hash_table_iter_t iter;
    ssn_hash_table_iter_init(&iter, engine->service_table);
    
    while (ssn_hash_table_iter_next(&iter) && result_count < max_results) {
        const char* key = (const char*)iter.key;
        
        // 过滤
        if (service_key[0] != '\0' && strncmp(key, service_key, strlen(service_key)) != 0) {
            continue;
        }
        
        ssn_service_info_t* service = (ssn_service_info_t*)iter.value;
        
        // 只返回可用的服务
        if (service->available) {
            memcpy(&results[result_count], service, sizeof(ssn_service_info_t));
            result_count++;
        }
    }
    
    ipc_mutex_unlock(engine->node_table_lock);
    
    LOG_DEBUG("Found %d services for type=%s, name=%s", result_count, service_type ? service_type : "*", 
             service_name ? service_name : "*");
    
    return result_count;
}
```

```c
// 加载发现配置
bool ssn_discovery_load_config(ssn_discovery_engine_t* engine, const char* config_file) {
    // 解析配置文件
    FILE* fp = fopen(config_file, "r");
    if (!fp) {
        LOG_ERROR("Failed to open discovery config file: %s", config_file);
        return false;
    }
    
    char line[256];
    while (fgets(line, sizeof(line), fp)) {
        // 跳过注释和空行
        if (line[0] == '#' || line[0] == '\n') {
            continue;
        }
        
        // 解析配置项
        char key[64], value[192];
        if (sscanf(line, "%63[^=]=%191[^\n]", key, value) == 2) {
            // 处理配置项
            if (strcmp(key, "multicast_group") == 0) {
                strncpy(engine->config.multicast_group, value, sizeof(engine->config.multicast_group) - 1);
            } else if (strcmp(key, "multicast_port") == 0) {
                engine->config.multicast_port = htons((uint16_t)atoi(value));
            } else if (strcmp(key, "heartbeat_interval") == 0) {
                engine->config.heartbeat_interval_ms = atoi(value);
            } else if (strcmp(key, "heartbeat_timeout") == 0) {
                engine->config.heartbeat_timeout_ms = atoi(value);
            }
        }
    }
    
    fclose(fp);
    LOG_INFO("Loaded discovery configuration from %s", config_file);
    return true;
}
```

```c
// 获取发现统计
bool ssn_discovery_get_stats(ssn_discovery_engine_t* engine, ssn_discovery_stats_t* stats) {
    if (!engine || !stats) {
        return false;
    }
    
    ipc_mutex_lock(engine->node_table_lock);
    
    stats->total_nodes = ssn_hash_table_size(engine->node_table);
    stats->active_nodes = 0;
    stats->offline_nodes = 0;
    
    // 统计各状态节点数
    ssn_hash_table_iter_t iter;
    ssn_hash_table_iter_init(&iter, engine->node_table);
    
    while (ssn_hash_table_iter_next(&iter)) {
        discovery_node_entry_t* entry = (discovery_node_entry_t*)iter.value;
        
        switch (entry->state) {
            case SSN_NODE_STATE_ACTIVE:
            case SSN_NODE_STATE_ANNOUNCED:
                stats->active_nodes++;
                break;
            case SSN_NODE_STATE_OFFLINE:
            case SSN_NODE_STATE_SUSPECTED:
                stats->offline_nodes++;
                break;
        }
    }
    
    stats->service_count = ssn_hash_table_size(engine->service_table);
    stats->announce_count = engine->stats.announce_count;
    stats->heartbeat_count = engine->stats.heartbeat_count;
    stats->query_count = engine->stats.query_count;
    
    ipc_mutex_unlock(engine->node_table_lock);
    
    return true;
}
```

## 9. 性能优化策略

### 9.1 心跳优化

1. **自适应心跳间隔**：根据网络状况动态调整心跳间隔
2. **批量心跳**：将多个节点的心跳合并发送
3. **差异更新**：只发送变化的节点信息

### 9.2 缓存策略

1. **本地缓存**：在本地缓存节点信息，减少网络请求
2. **TTL缓存**：为缓存项设置TTL，自动过期
3. **预取策略**：根据访问模式预取可能需要的节点信息

### 9.3 网络优化

1. **消息压缩**：对发现消息进行压缩传输
2. **合并更新**：将多个更新合并为一条消息
3. **优先级队列**：高优先级消息优先处理

## 10. 安全考虑

### 10.1 节点认证

1. **证书认证**：节点使用证书进行身份认证
2. **令牌认证**：使用预共享令牌进行快速认证
3. **白名单控制**：只允许白名单中的节点加入

### 10.2 消息安全

1. **消息签名**：对发现消息进行数字签名
2. **消息加密**：对敏感信息进行加密传输
3. **校验和验证**：验证消息完整性

### 10.3 访问控制

1. **节点权限**：不同节点有不同的发现权限
2. **服务可见性**：控制服务的可见范围
3. **查询限制**：限制查询频率和范围

## 11. 总结

节点发现机制为SSN提供了自动发现和动态管理分布式节点的能力。通过支持组播、目录服务和混合三种发现模式，能够适应不同规模和拓扑结构的分布式系统。完善的故障检测和恢复机制确保了系统的可靠性和容错性。
