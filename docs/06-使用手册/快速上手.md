# SSN 快速上手

> 目标读者：第一次接触 SSN（cd-ipc-ssn）的开发者。按本页步骤操作，5 分钟内即可跑通第一个 server + client 程序，并可直接套用三种常用场景模板。
> 面向版本：v2.5.1。文中所有 API 均对照真实头文件（`src/ssn_client.h`、`src/ssn_server.h`），与《使用指南》保持一致。

## 1. 5 分钟开始

### 1.1 环境要求

- 操作系统：Linux / POSIX（macOS 亦可）；Windows 不支持原生构建，请使用 WSL
- 构建工具：CMake >= 3.12、GCC >= 4.8（或 Clang >= 3.0）
- 依赖库：pthread；语言标准 C99

### 1.2 构建库

```bash
git clone https://github.com/acoinfo/cd-ipc-ssn.git
cd cd-ipc-ssn
mkdir -p build && cd build
cmake ..
make -j$(nproc)
```

构建产物为 `build/libssn_transport.so`（SOVERSION `libssn_transport.so.2`）。完整的安装与集成步骤见[部署手册](../05-部署手册/部署手册.md)。

### 1.3 第一个程序（server + client）

将下面两个文件保存到仓库根目录（与 `build/` 同级）。

**server.c —— 最小服务器**：

```c
#include <stdio.h>
#include <unistd.h>
#include "ssn_server.h"

// 消息回调：收到客户端消息时被调用
static void message_handler(ssn_server_t *server, ssn_peer_id_t id,
                            ssn_url_ref_t *url, ssn_data_ref_t *data, void *arg)
{
    (void)server; (void)url; (void)arg;
    printf("收到来自客户端 %u 的消息：%.*s\n",
           id, (int)data->length, (const char *)data->data);
}

int main(void)
{
    // 创建并启动服务器（Unix Socket 地址）
    ssn_server_t *server = ssn_server_create("unix:///tmp/hello_server");
    if (!server) { return 1; }
    ssn_server_set_message_handler(server, message_handler, NULL);
    if (!ssn_server_start(server)) { ssn_server_destroy(server); return 1; }

    // 事件循环驱动（必须）：服务器不会自动处理任何事件，
    // 连接握手、消息分发都由 ssn_server_poll 轮询触发
    int elapsed = 0;
    while (elapsed < 10) {          // poll 100ms + sleep 1s，共运行 10 秒
        ssn_server_poll(server, 100);
        sleep(1);
        elapsed++;
    }

    ssn_server_destroy(server);
    return 0;
}
```

**client.c —— 最小客户端**：

```c
#include <stdio.h>
#include <unistd.h>
#include "ssn_client.h"

int main(void)
{
    ssn_client_t *client = ssn_client_create();
    if (!client) { return 1; }

    // 连接服务器（同步，最多等待 5 秒）
    struct timespec timeout = { .tv_sec = 5, .tv_nsec = 0 };
    if (!ssn_client_connect(client, "unix:///tmp/hello_server", &timeout)) {
        printf("连接服务器失败\n");
        ssn_client_close(client);
        return 1;
    }

    // 发送消息（url_len 为 strlen 语义，不含结尾 NUL）
    ssn_url_ref_t url = { .url = "/hello", .url_len = 6 };
    ssn_data_ref_t data = { .data = "Hello, SSN!", .length = 11 };
    if (ssn_client_message(client, &url, &data) < 0) {
        printf("发送消息失败\n");
    }

    sleep(1);                       // 等待服务器处理
    ssn_client_close(client);
    return 0;
}
```

### 1.4 编译与运行验证

```bash
# 在仓库根目录编译（-I src 指向头文件，-L build 指向库）
gcc -std=c99 -o server server.c -I src -L build -lssn_transport -lpthread -Wl,-rpath,$PWD/build
gcc -std=c99 -o client client.c -I src -L build -lssn_transport -lpthread -Wl,-rpath,$PWD/build
```

打开两个终端分别运行：

```bash
# 终端 1：启动服务器（运行 10 秒后自动退出）
./server
# 终端 2：启动客户端
./client
```

预期结果：服务器终端打印 `收到来自客户端 0 的消息：Hello, SSN!`。若客户端先于服务器启动导致连接失败，稍后重新运行 `./client` 即可。

## 2. 常用场景模板

以下模板为可复制的最小骨架（为控制篇幅省略了错误处理细节，完整可运行版本见[示例代码](../../examples/)与[使用指南](使用指南.md)第 5 章）。

### 2.1 RPC 服务模板

**server.c —— 注册 `/math/add` 方法并进入事件循环**：

```c
#include <stdio.h>
#include <string.h>
#include "ssn_server.h"
#include "ssn_error.h"

static void add_handler(ssn_server_t *server, ssn_peer_id_t id,
                        ssn_header_t *hdr, ssn_url_ref_t *url,
                        ssn_data_ref_t *data, void *arg)
{
    (void)url; (void)arg;
    int a = 0, b = 0;
    sscanf((const char *)data->data, "{\"a\": %d, \"b\": %d}", &a, &b);
    char buf[32];
    int len = snprintf(buf, sizeof(buf), "%d", a + b);
    ssn_data_ref_t resp = { .data = buf, .length = (size_t)len };
    ssn_server_response(server, id, SSN_ECODE_SUCCESS, ssn_get_seqno(hdr), &resp);
}

int main(void)
{
    ssn_server_t *server = ssn_server_create("unix:///tmp/rpc_server");
    ssn_url_ref_t url = { .url = "/math/add", .url_len = 9 };
    ssn_server_add_method(server, &url, add_handler, NULL);
    ssn_server_start(server);
    while (1) {
        ssn_server_poll(server, 100);      // 事件循环：轮询驱动 RPC 请求
    }
}
```

**client.c —— 调用 RPC 并打印结果**：

```c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include "ssn_client.h"
#include "ssn_error.h"

static void reply_handler(ssn_client_t *client, ssn_header_t *hdr,
                          ssn_data_ref_t *data, void *arg)
{
    (void)client; (void)arg;
    if (hdr == NULL) { printf("RPC 超时\n"); return; }   // 服务器未响应
    if (ssn_get_status(hdr) != SSN_ECODE_SUCCESS) { return; }
    printf("结果：%.*s\n", (int)data->length, (const char *)data->data);
}

int main(void)
{
    ssn_client_t *client = ssn_client_create();
    struct timespec t = { .tv_sec = 5, .tv_nsec = 0 };
    ssn_client_connect(client, "unix:///tmp/rpc_server", &t);
    ssn_url_ref_t url = { .url = "/math/add", .url_len = 9 };
    ssn_data_ref_t in = { .data = "{\"a\": 5, \"b\": 3}", .length = 16 };
    ssn_client_call(client, &url, &in, reply_handler, NULL, 5000);
    /* 事件循环：应答回调由 poll 驱动，轮询直到回调触发（这里固定轮询 2 秒） */
    for (int i = 0; i < 20; i++) {
        ssn_client_poll(client, 100);
    }
    ssn_client_close(client);
    return 0;
}
```

### 2.2 PubSub 推送模板

**publisher.c —— 服务器周期向 `/news` 主题推送**：

```c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include "ssn_server.h"

int main(void)
{
    ssn_server_t *server = ssn_server_create("unix:///tmp/pubsub_server");
    ssn_server_start(server);
    ssn_url_ref_t url = { .url = "/news", .url_len = 5 };
    for (int i = 0; i < 5; i++) {
        ssn_server_poll(server, 100);      // 事件循环：先处理订阅握手
        char msg[64];
        int len = snprintf(msg, sizeof(msg), "新闻 %d", i + 1);
        ssn_data_ref_t data = { .data = msg, .length = (size_t)len };
        ssn_server_publish(server, &url, &data);
        ssn_server_poll(server, 100);      // 事件循环：分发消息
        sleep(1);
    }
    ssn_server_destroy(server);
    return 0;
}
```

**subscriber.c —— 订阅 `/news` 并轮询接收**：

```c
#include <stdio.h>
#include <unistd.h>
#include "ssn_client.h"

static void on_news(ssn_client_t *client, ssn_url_ref_t *url,
                    ssn_data_ref_t *data, void *arg)
{
    (void)client; (void)arg;
    /* url->url 非 NUL 结尾，用 %.*s + url_len 打印 */
    printf("收到 %.*s：%.*s\n", (int)url->url_len, url->url,
           (int)data->length, (const char *)data->data);
}

int main(void)
{
    ssn_client_t *client = ssn_client_create();
    ssn_client_set_on_message(client, on_news, NULL);
    struct timespec t = { .tv_sec = 5, .tv_nsec = 0 };
    ssn_client_connect(client, "unix:///tmp/pubsub_server", &t);
    ssn_url_ref_t url = { .url = "/news", .url_len = 5 };
    ssn_client_subscribe(client, &url, on_news, NULL, 5000);
    for (int i = 0; i < 10; i++) {
        ssn_client_poll(client, 1000);     // 事件循环：接收推送消息
    }
    ssn_client_close(client);
    return 0;
}
```

### 2.3 点对点消息模板

**server.c —— 收到消息后定向回复发送方**：

```c
#include <stdio.h>
#include <string.h>
#include "ssn_server.h"

static void on_msg(ssn_server_t *server, ssn_peer_id_t id,
                   ssn_url_ref_t *url, ssn_data_ref_t *data, void *arg)
{
    (void)url; (void)arg;
    printf("收到客户端 %u：%.*s\n", id, (int)data->length, (const char *)data->data);
    ssn_url_ref_t reply_url = { .url = "/reply", .url_len = 6 };
    ssn_data_ref_t reply = { .data = "已收到", .length = strlen("已收到") };
    ssn_server_message(server, id, &reply_url, &reply);  // 定向发给该客户端
}

int main(void)
{
    ssn_server_t *server = ssn_server_create("unix:///tmp/p2p_server");
    ssn_server_set_message_handler(server, on_msg, NULL);
    ssn_server_start(server);
    while (1) {
        ssn_server_poll(server, 100);      // 事件循环：处理消息分发
    }
}
```

**client.c —— 发送消息并接收定向回复**：

```c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include "ssn_client.h"

static void on_reply(ssn_client_t *client, ssn_url_ref_t *url,
                     ssn_data_ref_t *data, void *arg)
{
    (void)client; (void)url; (void)arg;
    printf("服务器回复：%.*s\n", (int)data->length, (const char *)data->data);
}

int main(void)
{
    ssn_client_t *client = ssn_client_create();
    ssn_client_set_on_message(client, on_reply, NULL);
    struct timespec t = { .tv_sec = 5, .tv_nsec = 0 };
    ssn_client_connect(client, "unix:///tmp/p2p_server", &t);
    ssn_url_ref_t url = { .url = "/hello", .url_len = 6 };
    ssn_data_ref_t data = { .data = "你好", .length = strlen("你好") };
    ssn_client_message(client, &url, &data);
    for (int i = 0; i < 5; i++) {
        ssn_client_poll(client, 1000);     // 事件循环：接收定向回复
    }
    ssn_client_close(client);
    return 0;
}
```

## 4. 常见错误速查

| 现象 | 原因与解决 |
|------|------------|
| `ssn_client_connect` 返回 false（连接失败） | 服务器未启动或已退出；地址格式错误（客户端地址必须带 `协议://` 前缀，如 `unix:///tmp/hello_server`）；连接超时太短。详见[FAQ](FAQ.md)连接类与《使用指南》8.1 |
| 服务器收不到客户端消息 | 事件循环未驱动：未调用 `ssn_server_poll`，连接握手与消息分发都不会发生。启动后必须循环 poll（见 1.3） |
| 客户端收不到订阅/回复消息 | 未调用 `ssn_client_poll` 接收，或未注册 `ssn_client_set_on_message` / 订阅回调 |
| `url_len` / `length` 填错 | 两者均为 strlen 语义（不含结尾 NUL 终止符），如 `url_len = 6` 对应 `/hello`；回调内指针指向的内容仅在回调返回前有效，需长期保存必须复制 |
| RPC 回调中 `ipc_hdr == NULL`（超时） | `ssn_client_call` 的 `timeout_ms` 过小、服务器未 poll、或方法未注册（返回 `SSN_ECODE_NOT_FOUND`，此时 `ipc_hdr` 非 NULL 且状态码非 0）。详见[FAQ](FAQ.md)超时类与《使用指南》8.2 |
| 编译链接报错 | 缺少 `-lssn_transport -lpthread`，或头文件/库路径不对（`-I src -L build`）；运行时需 rpath 或 `LD_LIBRARY_PATH=build` |
| 大消息发送失败 | 单包上限 128 KiB（`SSN_MAX_PACKET_SIZE`），超过需在应用层分片 |

完整故障排查见[FAQ](FAQ.md)与《使用指南》第 8 章「常见问题」。
